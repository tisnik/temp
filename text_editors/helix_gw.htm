3. Namapování příkazu gw na odlišnou klávesu

[keys.normal]
F1 = "goto_word"


theme = "catppuccin_latte"
 
[editor]
mouse = false
true-color = true
#end-of-line-diagnostics = "hint"
 
#[editor.inline-diagnostics]
#cursor-line = "warning"
#other-lines = "disable"
 
[editor.cursor-shape]
insert = "block"
normal = "block"
select = "block"
 
[keys.normal]
ins = "insert_mode"
esc = ["collapse_selection", "keep_primary_selection"]
F1 = "goto_word"
F2 = ":w"
F3 = "file_picker"
F4 = ":q"
F5 = ":sh python '%val{filename}'"
F6 = "rotate_view"
 
[keys.insert]
ins = "normal_mode"
 
[editor.statusline]
left = ["mode", "spinner", "version-control", "file-name"]
center = ["diagnostics", "file-encoding", "file-line-ending", "file-type"]
