4. Sjednocení typů

--!strict
 
local x: string = "foo"
 
x = 42
 
print(x)


--!strict
 
local x: string | number = "foo"
 
x = 42
 
print(x)


--!strict
 
function swap(x: number | string | boolean, y: number):(number, number | string | boolean)
    return y, x
end
 
print(swap(1, 2))
print(swap("foo", 2))
print(swap(true, 2))


--!strict
 
function swap(x: number | string | boolean, y: number):(number, number | string | boolean)
    return y, x
end
 
local x1: number = 0
local y1: number = 0
x1, y1 = swap(1, 2)
print(x1, y1)
 
local x2: number = 0
local y2: string = ""
x2, y2 = swap("foo", 2)
print(x2, y2)
 
local x3: number = 0
local y3: boolean = false
x3, y3 = swap(true, 2)
print(x3, y3)


--!strict
 
function foo(x: number | string | boolean):(number | string | boolean)
    return x
end
 
local x: number = foo(0)
local y: string = foo("foo")
local z: boolean = foo(true)
 
print(x, y, z)


--!strict
 
function foo(x: number | string | boolean):(number | string | boolean)
    return x
end
 
local x: number | string | boolean = foo(0)
print(type(x))
 
x = foo("foo")
print(type(x))
 
x = foo(true)
print(type(x))

5. Průnik typů

--!strict
 
type XCoord = {x: number}
type YCoord = {y: number}
 
type Vector2 = XCoord & YCoord
 
local vec2: Vector2 = {x = 1, y = 2}
 
print(vec2.x, vec2.y)


--!strict
 
type XCoord = {x: number}
type YCoord = {y: number}
type ZCoord = {z: number}
 
type Vector2 = XCoord & YCoord
type Vector3 = XCoord & YCoord & ZCoord
 
local vec2: Vector2 = {x = 1, y = 2}
local vec3: Vector3 = {x = 1, y = 2, z = 3}
 
print(vec2.x, vec2.y)
print(vec3.x, vec3.y, vec3.z)


--!strict
 
type XCoord = {x: number}
type YCoord = {y: number}
type ZCoord = {z: number}
 
type Vector2 = XCoord & YCoord
type Vector3 = XCoord & YCoord & ZCoord
 
local vec2: Vector2 = {x = 1, y = 2}
local vec3: Vector3 = {y = 2, z = 3}
 
print(vec2.x, vec2.y)
print(vec3.x, vec3.y, vec3.z)


--!strict
 
type Coord1 = {x: number, y: number}
type Coord2 = {y: number, z: number}
 
type Vector3 = Coord1 & Coord2
 
local vec3: Vector3 = {x = 1, y = 2, z = 3}
 
print(vec3.x, vec3.y, vec3.z)

6. Specifikace typu „funkce“ u proměnných

function zero(x: number): number
    return 0
end
 
function inc(x: number): number
    return x+1
end
 
function dec(x: number): number
    return x-1
end
 
local x: (number) -> number
 
x = zero
print(x(42))
 
x = inc

x(42)

8. Nepovinné parametry funkcí

--!strict

function add(x: number, y: number, z: number?): number
    if z == nil then
        return x + y
    else
        return x + y + z
    end
end

print(add(1, 2))
print(add(1, 2, 3))
print(add(1, 2, 3, 4))

11. Funkce odpovídající zapsanému typu

--!strict

type math_function = (number) -> number

function zero(x: number): number
    return 0
end

function inc(x: number): number
    return x+1
end

function dec(x: number): number
    return x-1
end

local x: math_function

x = zero
print(x(42))

x = inc
print(x(42))

x = dec
print(x(42))


--!strict

type login_implementation = (number | string) -> boolean

function login_by_id(id: number): boolean
    return id == 0
end

function login_by_name(name: string): boolean
    return name == "root"
end

local login: login_implementation

login = login_by_id
print(login(0))
print(login(1))

login = login_by_name
print(login("root"))
print(login("admin"))
