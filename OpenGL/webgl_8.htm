function makeTexture(width, height) {
    const array = new Array(width*height*4);
    const divider = width / 8;
 
    let index = 0;
    for (j=0; j<height; j++) {
        for (i=0; i<width; i++) {
            const alpha = ((i&divider)==0^(j&divider)==0) * 255;
            array[index++] = 128;
            array[index++] = 256*i/width;
            array[index++] = 256*j/width;
            array[index++] = alpha;            // alfa slozka
        }
    }
    return array;
}

    const blending_func_1 = gl[blending_func_1_value];
    const blending_func_2 = gl[blending_func_2_value];
 
    // barva na kterou se vymaže pozadí před vykreslením textury
    const r = document.getElementById("r").value;
    const g = document.getElementById("g").value;
    const b = document.getElementById("b").value;
    const a = document.getElementById("a").value;

    if (enable_blending) {
        gl.enable(gl.BLEND);
    } else {
        gl.disable(gl.BLEND);

    // nastavení blendingu
    gl.blendFunc(blending_func_1, blending_func_2);
 
    // navázání na atribut ve shaderech
    const samplerLoc = gl.getUniformLocation(program, "texture");
    gl.uniform1i(samplerLoc, 0);
 
    // transformace souřadnic: z normalizovaných souřadnic na souřadnice na plátnu
    gl.viewport(0, 0, canvas.width, canvas.height);
 
    // nastavení barvy pro vymazání barvového bufferu
    gl.clearColor(r, g, b, a);
 
    // vymazání barvového bufferu
    gl.clear(gl.COLOR_BUFFER_BIT);
 
    // spuštění programu
    gl.useProgram(program);
 
    // vykreslení dvojice trojúhelníků
    gl.drawArrays(gl.TRIANGLES, 0, 6);
}

function createTexture(alpha) {
    // textura (prozatím primitivní)
    const texture = gl.createTexture();
    gl.bindTexture(gl.TEXTURE_2D, texture);
    gl.pixelStorei(gl.UNPACK_FLIP_Y_WEBGL, true);

    const textureWidth = 16
    const textureHeight = 16

    // textura
    textureArray = makeTexture(textureWidth, textureHeight, alpha);

function makeTexture(width, height) {
    const array = new Array(width*height*4);
    const divider = width / 8;
 
    let index = 0;
    for (j=0; j<height; j++) {
        for (i=0; i<width; i++) {
            const alpha = ((i&divider)==0^(j&divider)==0) * 255;
            array[index++] = 128;

function matrix_identity() {

function matrix_rotate_around_x(a) {
    const c = Math.cos(a);
    const s = Math.sin(a);
    return new Float32Array([
        1, 0, 0, 0,
        0, c, s, 0,
        0, -s, c, 0,
        0, 0, 0, 1
    ]);
}

    const c = Math.cos(a);
    const s = Math.sin(a);
    return new Float32Array([
        c, 0, -s, 0,
        0, 1, 0, 0,
        s, 0, c, 0,
        0, 0, 0, 1
    ]);

function matrix_perspective_fov(fovy, aspect, near, far) {
    const f = 1.0 / Math.tan(fovy / 2);
    const nf = 1 / (near - far);
    return new Float32Array([
        f / aspect, 0, 0, 0,
        0, f, 0, 0,
        0, 0, (far + near) * nf, -1,
        0, 0, (2 * far * near) * nf, 0
    ]);
}
 
function matrix_multiply(a, b) {
    const out = new Float32Array(16);
    for (let col = 0; col < 4; col++) {
        for (let row = 0; row < 4; row++) {
            out[col * 4 + row] =
              a[0 * 4 + row] * b[col * 4 + 0] +
              a[1 * 4 + row] * b[col * 4 + 1] +
              a[2 * 4 + row] * b[col * 4 + 2] +
              a[3 * 4 + row] * b[col * 4 + 3];
        }
    }
    return out;
}

18. Obsah navazujícího článku: plně programovatelný vykreslovací řetězec

Prozatím jsme se v tomto seriálu zabývali především popisem standardizovaného aplikačního rozhraní (API) nabízeného Open GL ES. Víme již, jakým postupem se provádí konfigurace vykreslovacího řetězce (vymazání bufferů, povolení blendingu atd.) a ukázali jsme si i způsoby předávání dat do vykreslovacího řetězce (to se provádí přes buffery, do nichž se posílají obsahy polí). Ovšem OpenGL ES nabízí v porovnání s původním „klasickým“ Open GL ještě jednu důležitou vlastnost: celý vykreslovací řetězec (rendering pipeline) není pouze konfigurovatelný, ale je i programovatelný. V této oblasti se využívají takzvané shadery, které jsme sice ve všech demonstračních příkladech používali (resp. jsme je dokonce byli nuceni používat), ovšem prozatím jsme se nezabývali tím, jak lze shadery využít pro realizaci různých grafických efektů. Ty se navíc spouští na GPU a jsou do značné míry nezávislé na výkonu hlavního mikroprocesoru. A právě tímto velmi důležitým tématem se budeme zabývat v navazující části tohoto seriálu.
